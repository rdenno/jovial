interface GanymedeApi {
    @GET("/api/v1/vod/pagination")
    suspend fun getVods(
        @Query("limit") limit: Int = 50,
        @Query("offset") offset: Int = 0
    ): VodPaginationResponse

    // Fetches initial messages right before a seek timestamp
    @GET("/api/v1/vod/{id}/chat/seek")
    suspend fun getSeekChat(
        @Path("id") vodId: String,
        @Query("time") timeSeconds: Double,
        @Query("count") count: Int = 30
    ): List<ChatComment>

    // Fetches chat comments between [start, end] seconds
    @GET("/api/v1/vod/{id}/chat")
    suspend fun getChatRange(
        @Path("id") vodId: String,
        @Query("start") startSeconds: Double,
        @Query("end") endSeconds: Double
    ): List<ChatComment>

    companion object {
        fun create(baseUrl: String, apiKey: String? = null): GanymedeApi {
            val client = OkHttpClient.Builder().apply {
                if (!apiKey.isNullOrBlank()) {
                    addInterceptor { chain ->
                        val req = chain.request().newBuilder()
                            .addHeader("X-API-Key", apiKey)
                            .build()
                        chain.proceed(req)
                    }
                }
            }.build()

            return Retrofit.Builder()
                .baseUrl(baseUrl.trimEnd('/') + "/")
                .client(client)
                .addConverterFactory(GsonConverterFactory.create())
                .build()
                .create(GanymedeApi::class.java)
        }
    }
}

class PlayerViewModel : ViewModel() {
    private var api: GanymedeApi? = null
    var serverUrl: String = "http://192.168.1.100:4000"
        private set

    private val _vods = MutableStateFlow<List<GanymedeVod>>(emptyList())
    val vods: StateFlow<List<GanymedeVod>> = _vods.asStateFlow()

    private val _layoutConfig = MutableStateFlow(PlayerLayoutConfig())
    val layoutConfig: StateFlow<PlayerLayoutConfig> = _layoutConfig.asStateFlow()

    private val _chatMessages = MutableStateFlow<List<ChatComment>>(emptyList())
    val chatMessages: StateFlow<List<ChatComment>> = _chatMessages.asStateFlow()

    private var lastSyncedSecond: Double = -1.0
    private var chatJob: Job? = null

    fun connectToServer(url: String, apiKey: String?) {
        serverUrl = url.trimEnd('/')
        api = GanymedeApi.create(serverUrl, apiKey)
        fetchVods()
    }

    fun fetchVods() {
        val currentApi = api ?: return
        viewModelScope.launch {
            runCatching { currentApi.getVods() }
                .onSuccess { _vods.value = it.data }
        }
    }

    fun updateLayout(transform: (PlayerLayoutConfig) -> PlayerLayoutConfig) {
        _layoutConfig.update(transform)
    }

    /**
     * Called every 500ms with the ExoPlayer's current playback timestamp in seconds.
     * Automatically detects scrubbing/seeking vs normal playback.
     */
    fun onPlaybackPositionUpdate(vodId: String, currentSeconds: Double) {
        if (_layoutConfig.value.displayMode == ChatDisplayMode.HIDDEN) return
        val currentApi = api ?: return

        // Detect seek (jump > 4 seconds backward or forward)
        if (lastSyncedSecond < 0 || abs(currentSeconds - lastSyncedSecond) > 4.0) {
            lastSyncedSecond = currentSeconds
            chatJob?.cancel()
            chatJob = viewModelScope.launch {
                runCatching {
                    currentApi.getSeekChat(vodId, currentSeconds, count = 40)
                }.onSuccess { initialBatch ->
                    _chatMessages.value = initialBatch
                }
            }
            return
        }

        // Normal forward playback: fetch window [lastSyncedSecond, currentSeconds]
        if (currentSeconds > lastSyncedSecond) {
            val start = lastSyncedSecond
            val end = currentSeconds
            lastSyncedSecond = currentSeconds

            viewModelScope.launch {
                runCatching {
                    currentApi.getChatRange(vodId, start, end)
                }.onSuccess { newComments ->
                    if (newComments.isNotEmpty()) {
                        _chatMessages.update { existing ->
                            (existing + newComments)
                                .distinctBy { it.id }
                                .takeLast(150) // Keep memory footprint tiny
                        }
                    }
                }
            }
        }
    }

    fun resetChatState() {
        lastSyncedSecond = -1.0
        _chatMessages.value = emptyList()
    }
}

@Composable
fun GanymedePlayerScreen(
    vod: GanymedeVod,
    viewModel: PlayerViewModel,
    onBack: () -> Unit
) {
    val context = LocalContext.current
    val config by viewModel.layoutConfig.collectAsState()
    val messages by viewModel.chatMessages.collectAsState()
    var showSettingsSheet by remember { mutableStateOf(false) }

    // Initialize ExoPlayer
    val exoPlayer = remember {
        ExoPlayer.Builder(context).build()
    }

    // Load MediaItem when VOD or video source preference (Raw vs Rendered Chat MP4) changes
    LaunchedEffect(vod.id, config.useRenderedChatVideo) {
        val currentPos = exoPlayer.currentPosition
        val videoUrl = vod.getFullVideoUrl(viewModel.serverUrl, config.useRenderedChatVideo)
        exoPlayer.setMediaItem(MediaItem.fromUri(videoUrl), currentPos)
        exoPlayer.prepare()
        exoPlayer.playWhenReady = true
    }

    // Poll ExoPlayer position every 500ms to sync chat
    LaunchedEffect(vod.id, config.displayMode) {
        while (true) {
            if (exoPlayer.isPlaying) {
                val currentSec = exoPlayer.currentPosition / 1000.0
                viewModel.onPlaybackPositionUpdate(vod.id, currentSec)
            }
            delay(500L)
        }
    }

    DisposableEffect(Unit) {
        onDispose {
            viewModel.resetChatState()
            exoPlayer.release()
        }
    }

    Box(modifier = Modifier.fillMaxSize().background(Color.Black)) {
        when (config.displayMode) {
            ChatDisplayMode.HIDDEN -> {
                VideoSurface(exoPlayer, Modifier.fillMaxSize())
            }

            ChatDisplayMode.SIDE_BY_SIDE -> {
                Row(modifier = Modifier.fillMaxSize()) {
                    if (config.chatSide == ChatSide.LEFT) {
                        ChatPane(
                            messages = messages,
                            fontSizeSp = config.fontSizeSp,
                            modifier = Modifier
                                .fillMaxHeight()
                                .fillMaxWidth(config.chatWidthFraction)
                                .background(Color(0xFF121216))
                        )
                        VideoSurface(exoPlayer, Modifier.fillMaxHeight().weight(1f))
                    } else {
                        VideoSurface(exoPlayer, Modifier.fillMaxHeight().weight(1f))
                        ChatPane(
                            messages = messages,
                            fontSizeSp = config.fontSizeSp,
                            modifier = Modifier
                                .fillMaxHeight()
                                .fillMaxWidth(config.chatWidthFraction)
                                .background(Color(0xFF121216))
                        )
                    }
                }
            }

            ChatDisplayMode.OVERLAY -> {
                Box(modifier = Modifier.fillMaxSize()) {
                    VideoSurface(exoPlayer, Modifier.fillMaxSize())

                    val alignment = if (config.chatSide == ChatSide.LEFT) {
                        Alignment.CenterStart
                    } else {
                        Alignment.CenterEnd
                    }

                    ChatPane(
                        messages = messages,
                        fontSizeSp = config.fontSizeSp,
                        modifier = Modifier
                            .align(alignment)
                            .fillMaxHeight(0.85f)
                            .fillMaxWidth(config.chatWidthFraction)
                            .padding(12.dp)
                            .background(
                                Color.Black.copy(alpha = config.overlayOpacity),
                                shape = RoundedCornerShape(8.dp)
                            )
                    )
                }
            }
        }

        // Top Floating Quick-Controls Bar (Back, Quick Side-Swap, Layout Settings)
        Row(
            modifier = Modifier
                .align(Alignment.TopStart)
                .padding(12.dp)
                .background(Color.Black.copy(alpha = 0.6f), RoundedCornerShape(24.dp))
                .padding(horizontal = 8.dp, vertical = 4.dp),
            verticalAlignment = Alignment.CenterVertically
        ) {
            IconButton(onClick = onBack) {
                Icon(Icons.Default.ArrowBack, contentDescription = "Back", tint = Color.White)
            }
            // One-tap button to swap Chat Left/Right
            if (config.displayMode != ChatDisplayMode.HIDDEN) {
                IconButton(
                    onClick = {
                        viewModel.updateLayout {
                            it.copy(
                                chatSide = if (it.chatSide == ChatSide.LEFT) ChatSide.RIGHT else ChatSide.LEFT
                            )
                        }
                    }
                ) {
                    Icon(Icons.Default.SwapHoriz, contentDescription = "Swap Chat Side", tint = Color.White)
                }
            }
            // Open full layout customization sheet
            IconButton(onClick = { showSettingsSheet = !showSettingsSheet }) {
                Icon(Icons.Default.Tune, contentDescription = "Customize Layout", tint = Color.White)
            }
        }

        // Customization Overlay Panel
        AnimatedVisibility(
            visible = showSettingsSheet,
            modifier = Modifier.align(Alignment.Center)
        ) {
            LayoutCustomizationCard(
                config = config,
                hasRenderedChatVideo = !vod.chatVideoPath.isNullOrBlank(),
                onUpdate = { viewModel.updateLayout(it) },
                onClose = { showSettingsSheet = false }
            )
        }
    }

@Composable
fun ChatPane(
    messages: List<ChatComment>,
    fontSizeSp: Float,
    modifier: Modifier = Modifier
) {
    val listState = rememberLazyListState()

    // Auto-scroll to bottom when new messages arrive
    LaunchedEffect(messages.size) {
        if (messages.isNotEmpty()) {
            listState.animateScrollToItem(messages.size - 1)
        }
    }

    LazyColumn(
        state = listState,
        modifier = modifier.padding(horizontal = 8.dp, vertical = 6.dp),
        verticalArrangement = Arrangement.spacedBy(4.dp)
    ) {
        items(messages, key = { it.id }) { comment ->
            val usernameColor = remember(comment.message.userColor) {
                try {
                    Color(AndroidColor.parseColor(comment.message.userColor ?: "#9146FF"))
                } catch (e: Exception) {
                    Color(0xFF9146FF)
                }
            }

            Text(
                text = buildAnnotatedString {
                    withStyle(SpanStyle(color = usernameColor, fontWeight = FontWeight.Bold)) {
                        append("${comment.commenter.displayName}: ")
                    }
                    withStyle(SpanStyle(color = Color.White)) {
                        append(comment.message.body)
                    }
                },
                fontSize = fontSizeSp.sp,
                lineHeight = (fontSizeSp + 4).sp
            )
        }
    }
}
